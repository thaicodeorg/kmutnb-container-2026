Workshop: Nginx Web Server

Building a Custom Nginx Image

KTU - Computer Science

Project Structure

nginx-centos/
├── Dockerfile
├── html/
│   └── index.html
├── conf/
│   └── nginx.conf
└── certs/
    ├── server.crt
    └── server.key

Lab 1: Create Dockerfile

FROM quay.io/centos/centos:stream9

ARG NGINX_VERSION=1.24.0

RUN dnf install -y nginx openssl openssl-devel \
    && dnf clean all

RUN useradd -r -s /bin/false nginx

RUN mkdir -p /var/cache/nginx \
    && chown -R nginx:nginx /var/cache/nginx

COPY conf/nginx.conf /etc/nginx/nginx.conf
COPY html/ /usr/share/nginx/html/

EXPOSE 80 443

CMD ["nginx", "-g", "daemon off;"]

Lab 2: Custom Nginx Config

worker_processes auto;
error_log /var/log/nginx/error.log;
pid /run/nginx.pid;

events {
    worker_connections 1024;
}

http {
    server {
        listen 80;
        server_name localhost;
        root /usr/share/nginx/html;
        index index.html;
    }
}

Lab 3: Build and Run

# Build the image
docker build -t my-nginx:centos .

# Run the container
docker run -d --name nginx-web \
    -p 8080:80 \
    my-nginx:centos

# Test
curl http://localhost:8080

# View logs
docker logs nginx-web

Lab 4: SSL Setup

# Generate self-signed certificate
openssl req -x509 -nodes -days 365 \
    -newkey rsa:2048 \
    -keyout certs/server.key \
    -out certs/server.crt \
    -subj "/CN=localhost"

# Update nginx.conf for HTTPS
server {
    listen 443 ssl;
    ssl_certificate /etc/nginx/certs/server.crt;
    ssl_certificate_key /etc/nginx/certs/server.key;
}

Cleanup

# Stop and remove container
docker stop nginx-web
docker rm nginx-web

# Remove image
docker rmi my-nginx:centos

# Remove unused images
docker image prune -a
